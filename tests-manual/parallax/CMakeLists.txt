cmake_minimum_required(VERSION 3.16)
project(ParallaxSmoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_AUTOMOC ON)
set(CLIENT_COMPONENT "Client")
find_package(Qt5 5.15 REQUIRED COMPONENTS Widgets Multimedia)
find_path(GLM_INCLUDE_DIR glm/glm.hpp REQUIRED)
add_executable(parallax-smoke main.cpp ../../interface/src/DesktopParallax.cpp
    ../../interface/src/NativeCameraTracker.cpp ../../interface/src/NativeHeadPose.cpp)
target_include_directories(parallax-smoke PRIVATE ../../interface/src ${GLM_INCLUDE_DIR})
target_link_libraries(parallax-smoke Qt5::Widgets)
include(../../cmake/NativeCamera.cmake)
target_native_camera(parallax-smoke)
enable_testing()
add_test(NAME native-parallax COMMAND parallax-smoke)
